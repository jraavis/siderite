  code={`use siderite::prelude::*;

#[derive(Deserialize, Validate, Schema)]
struct Greeting { shout: Option<bool> }

/// Greet somebody by name.
#[get("/hello/{name}", tag = "greetings")]
async fn hello(Path(name): Path<String>, Query(q): Query<Greeting>) -> PlainText<String> {
    let msg = format!("hello, {name}");
    PlainText(if q.shout.unwrap_or(false) { msg.to_uppercase() } else { msg })
}

#[tokio::main]
async fn main() -> Result<(), ServerError> {
    App::new().title("Hello").routes(routes![hello]).run("127.0.0.1:8000").await
}`}

<div class="section-label">What is in the box</div>

<dl class="feature-list">
  <div>
    <dt>HTTP</dt>
    <dd>Route macros, extractors, middleware, lifespan, WebSockets, and RFC 7807 errors. OpenAPI 3.1 is generated from handler signatures and served at <code>/docs</code>.</dd>
  </div>
  <div>
    <dt>Validation</dt>
    <dd>A Pydantic-style pipeline on <code>#[derive(Validate, Schema)]</code>. Invalid input is a 422 that lists every error with its location.</dd>
  </div>
  <div>
    <dt>ORM</dt>
    <dd><code>#[derive(Model)]</code> and lazy <code>QuerySet</code>s. <code>User::age.icontains(..)</code> does not compile. Unsupported backend features fail with a capability error before any I/O.</dd>
  </div>
  <div>
    <dt>Production</dt>
    <dd>Layered configuration, signals, database routing, cache, CLI, testkit, and migrations for SQLite, PostgreSQL, and MySQL.</dd>
  </div>
</dl>
